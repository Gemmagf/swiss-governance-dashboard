Example 7: Dynamic Content in JavaScript

How to replace hardcoded strings in your main script

BEFORE:

// Chart labels
const chartTitle = "Rànquing cantonal";
const scenarioLabel = "Tendencial";
const yAxisLabel = m.unit;

// Messages
const accuracyMsg = "Fiabilitat del model: " + mape + "%";
const costMsg = "Cost anual estimat: " + cost + " mio";

// Status text
const statusYear = year > YOBS ? "previsió" : "observat";
const statusText = "Dada " + statusYear;

AFTER:

// Chart labels
const chartTitle = t("section.ranking");
const scenarioLabel = t("ctrl.scenario_baseline");
const yAxisLabel = m.unit;

// Messages
const accuracyMsg = t("model.accuracy_short", { pct: mape.toFixed(1) });
const costMsg = t("chart.annual_cost") + ": " + cost + " mio";

// Status text
const statusKey = year > YOBS ? "ctrl.year_status_forecast" : "ctrl.year_status_observed";
const statusText = t(statusKey);

// Handle language changes for charts
window.addEventListener("languageChanged", () => {
  updateChartLabels();
  refreshSimulatorUI();
  redrawAlerts();
});
